cmake_minimum_required(VERSION 3.24)
project(live_peer VERSION 0.1.0 LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(FetchContent)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(LIVE_PEER_BUILD_EXAMPLES "Build the local interoperability driver" ${PROJECT_IS_TOP_LEVEL})
option(LIVE_PEER_SANITIZE "Instrument library and dependencies with ASan/UBSan" ON)
if(LIVE_PEER_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
option(LIVE_PEER_SYSTEM_MBEDTLS "Use an existing Mbed TLS >=3.6.7, <4 installation" OFF)
if(LIVE_PEER_SYSTEM_MBEDTLS)
  find_package(MbedTLS 3.6.7 REQUIRED CONFIG)
else()
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(mbedtls
    URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.7/mbedtls-3.6.7.tar.bz2
    URL_HASH SHA256=a7e8bcbec0e6f761b4af24f25677626b35f762f68eef79c08677a363212d11f6)
  FetchContent_MakeAvailable(mbedtls)
  foreach(t mbedtls mbedx509 mbedcrypto)
    target_compile_definitions(${t} PUBLIC MBEDTLS_SSL_DTLS_SRTP)
  endforeach()
endif()
find_package(Threads REQUIRED)
set(LIBSRTP_TEST_APPS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENSSL OFF CACHE BOOL "" FORCE)
set(sctp_build_programs OFF CACHE BOOL "" FORCE)
set(sctp_build_shared_lib OFF CACHE BOOL "" FORCE)
set(sctp_werror OFF CACHE BOOL "" FORCE)
FetchContent_Declare(srtp
  URL https://codeload.github.com/cisco/libsrtp/tar.gz/6ff02afa8d2dc3f2e8896af391d137c1c66ff6fa
  URL_HASH SHA256=1ef3e6315e0ea3cd05f4980c8b5f7a9df369e3d586f5e0616e6cc4c778401d4c)
FetchContent_Declare(usrsctp
  URL https://codeload.github.com/sctplab/usrsctp/tar.gz/c625e21950600460cb5c6075616eb1622df2b06a
  URL_HASH SHA256=090b9d03bf068ce5989576c8e0ff6979d3031ea85e405acd263206f19166b41e)
FetchContent_MakeAvailable(srtp usrsctp)
set(SCTP_SOURCE_DIR "${usrsctp_SOURCE_DIR}")
include("${CMAKE_CURRENT_SOURCE_DIR}/patches/usrsctp-reinit.cmake")
# Upstream libsrtp's build-tree include layout differs from its installed layout.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/srtp2")
configure_file("${srtp_SOURCE_DIR}/include/srtp.h"
  "${CMAKE_CURRENT_BINARY_DIR}/include/srtp2/srtp.h" COPYONLY)
add_library(live_peer STATIC
  src/address.c src/agent.c src/dtls_srtp.c src/ice.c
  src/peer.c src/peer_connection.c src/ports.c src/rtcp.c
  src/rtp.c src/sctp.c src/sdp.c src/socket.c src/stun.c src/utils.c)
add_library(live_peer::peer ALIAS live_peer)
option(LIVE_PEER_COVERAGE "Collect Clang source coverage for the local library" OFF)
if(LIVE_PEER_COVERAGE)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "LIVE_PEER_COVERAGE requires Clang")
  endif()
  target_compile_options(live_peer PRIVATE -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(live_peer INTERFACE -fprofile-instr-generate)
endif()
target_include_directories(live_peer PUBLIC include PRIVATE src
  "${CMAKE_CURRENT_BINARY_DIR}/include" "${usrsctp_SOURCE_DIR}/usrsctplib")
target_compile_definitions(live_peer PRIVATE CONFIG_DTLS_USE_ECDSA=1 CONFIG_USE_USRSCTP=1)
target_compile_options(live_peer PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(live_peer PRIVATE MbedTLS::mbedtls MbedTLS::mbedx509
  MbedTLS::mbedcrypto srtp2 usrsctp Threads::Threads)
if(LIVE_PEER_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_link_options(live_peer INTERFACE -fsanitize=address,undefined)
endif()
if(LIVE_PEER_BUILD_EXAMPLES)
  add_executable(peer_interop examples/interop.c)
  target_link_libraries(peer_interop PRIVATE live_peer)
endif()
if(BUILD_TESTING)
  add_executable(peer_security_tests tests/security.c)
  target_include_directories(peer_security_tests PRIVATE src
    "${CMAKE_CURRENT_BINARY_DIR}/include" "${usrsctp_SOURCE_DIR}/usrsctplib")
  target_link_libraries(peer_security_tests PRIVATE live_peer MbedTLS::mbedtls
    MbedTLS::mbedx509 MbedTLS::mbedcrypto srtp2)
  add_test(NAME peer_security COMMAND peer_security_tests)
  set_tests_properties(peer_security PROPERTIES TIMEOUT 30)
endif()
option(LIVE_PEER_FUZZ "Build libFuzzer parser harness (Clang)" OFF)
if(LIVE_PEER_FUZZ)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "LIVE_PEER_FUZZ requires Clang")
  endif()
  foreach(t live_peer mbedtls mbedx509 mbedcrypto srtp2 usrsctp)
    if(TARGET ${t})
      get_target_property(imported ${t} IMPORTED)
      if(NOT imported)
        target_compile_options(${t} PRIVATE -fsanitize=fuzzer-no-link)
      endif()
    endif()
  endforeach()
  add_executable(peer_fuzz tests/fuzz.c)
  add_executable(peer_fuzz_dtls tests/fuzz_dtls.c)
  add_executable(peer_fuzz_sctp tests/fuzz_sctp.c)
  foreach(t peer_fuzz peer_fuzz_dtls peer_fuzz_sctp)
    target_include_directories(${t} PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/include" "${usrsctp_SOURCE_DIR}/usrsctplib")
    target_compile_options(${t} PRIVATE -fsanitize=fuzzer)
    target_link_options(${t} PRIVATE -fsanitize=fuzzer)
    target_link_libraries(${t} PRIVATE live_peer MbedTLS::mbedtls MbedTLS::mbedx509 MbedTLS::mbedcrypto srtp2)
  endforeach()
endif()
